  \fill (0,0) circle (1.4pt) node[below=2pt] {$i$};
  \fill (1.75,0) circle (1.4pt) node[below=2pt] {$j$};
  \draw[gray,thin] (.75,0) -- (.75,1.3);
  \draw[gray,thin] (1,0) -- (1,1.3);
  \draw[accent,very thick,|-|] (.75,1.3) -- (1,1.3) node[midway,above] {$\xi_{ij}$};
  \draw[->,very thick] (0,0) -- (-1.4,0) node[left] {$\vect F_{ij}$};
  \draw[->,thick,itbablue] (-.1,.45) -- (.55,.45) node[midway,above] {$\hat{\vect e}_{ij}$};
\end{tikzpicture}
\end{column}
\begin{column}{.58\textwidth}
\textbf{Sistema real:} discos rígidos que se deforman levemente al chocar, como un medio granular.
\medskip

\textbf{Ecuación de movimiento}
\[m\,\ddot{\vect r}_i=\vect F_i=\sum_{j:\ \xi_{ij}>0}-k\,\xi_{ij}\,\hat{\vect e}_{ij}\]
\[\xi_{ij}=r_i+r_j-|\vect r_j-\vect r_i|,\qquad \hat{\vect e}_{ij}=\frac{\vect r_j-\vect r_i}{|\vect r_j-\vect r_i|}\]
$j$: otra partícula o un cuerpo fijo de masa infinita.
\medskip

\textbf{Energía} (sin fricción ni disipación, se conserva)
\[E=\sum_i\tfrac12 m\,|\vect v_i|^2+\sum_{\text{contactos}}\tfrac12 k\,\xi^2\]
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Integración con paso temporal fijo: Verlet original}
\begin{columns}[c,onlytextwidth]
\begin{column}{.55\textwidth}
\[\vect r(t+\dt)=2\vect r(t)-\vect r(t-\dt)+\frac{\dt^2}{m}\vect F(t)\]
\[\vect v(t)=\frac{\vect r(t+\dt)-\vect r(t-\dt)}{2\dt}\]
\textbf{Arranque} (Euler hacia atrás):
\[\vect r(-\dt)=\vect r(0)-\dt\,\vect v(0)+\frac{\dt^2}{2m}\vect F(0)\]
\end{column}
\begin{column}{.43\textwidth}
\begin{itemize}\setlength\itemsep{.25cm}
\item Una evaluación de fuerzas por paso.
\item $\vect v(t)$ recién se conoce al calcular $\vect r(t+\dt)$: el estado en $t$ se escribe en ese momento.
\item Se guarda cada $n$ pasos: $\dt_2=n\,\dt$.
\item El motor solo escribe estado; los observables se calculan aparte.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Algoritmo del motor}
\begin{columns}[c,onlytextwidth]
\begin{column}{.62\textwidth}
\begin{block}{Billar}
\small
\begin{tabbing}
xx\=xx\=\kill
Generar $N$ partículas azules sin superposición, $|\vect v|=v_0$\\
Calcular $\vect F(0)$ y $\vect r(-\dt)$\\
\textbf{para} $n=0,1,\dots,t_f/\dt$:\\
\> Ubicar las partículas en celdas de lado $\geq 2r$\\
\> $\vect F\leftarrow$ resortes con partículas vecinas,\\
\> \> pared (imagen) y obstáculos\\
\> Si toca un obstáculo: azul $\to$ roja\\
\> $\vect r(t+\dt)$ y $\vect v(t)$ con Verlet\\
\> \textbf{si} $n$ es múltiplo de $\dt_2/\dt$: escribir $t$, $\vect r$, $\vect v$ y color
\end{tabbing}
\end{block}
\end{column}
\begin{column}{.36\textwidth}
\begin{itemize}\setlength\itemsep{.25cm}
\item Con celdas, cada partícula se compara solo con su celda y las 8 vecinas: costo proporcional a $N$.
\item Cada par se recorre una vez: $\vect F_{ji}=-\vect F_{ij}$.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Observables}
\small
\begin{columns}[T,onlytextwidth]
\begin{column}{.49\textwidth}
\textbf{Energía} (cada cuadro, contando cada resorte una vez)
\[\delta E=\max_{t\le t_f}\frac{|E(t)-E(0)|}{E(0)}\]
\textbf{Conversión} (por realización)
\[\fu(t)=\frac{N_u(t)}{N}\]
$t_{90}$, $t_{100}$: primer cuadro con $\fu\geq0{,}9$ y $\fu=1$.
\end{column}
\begin{column}{.49\textwidth}
\textbf{Promedios entre las $M$ realizaciones}
\[\langle t_{90}\rangle=\frac1M\sum_{k=1}^{M}t_{90}^{(k)}\]
\[\sigma^2=\frac{1}{M-1}\sum_{k=1}^{M}\big(t_{90}^{(k)}-\langle t_{90}\rangle\big)^2\]
Si alguna realización no llega, se informa cuántas y $\fu(t_f)$.
\medskip

\textbf{Rapidez}: $f(v)$ = fracción de partículas con $|\vect v|$ en el bin / ancho del bin ($0{,}1$ m/s).
\medskip

\textbf{Tiempo de ejecución}: reloj de la integración, sin generación ni escritura.
\end{column}

\section{Conclusiones}
\begin{frame}\sectionpage\end{frame}

\begin{frame}{Conclusiones}
\small
\begin{itemize}\setlength\itemsep{.15cm}
\item \textbf{Integradores:} Beeman da el menor error, $\approx13\,\%$ menos que Verlet, con el mismo orden 2; Euler predictor-corrector es de orden 1.
\item \textbf{Paso:} $\delta E$ crece con $\dt$ y con el tiempo; $\dt=\pasoElegido$ s mantiene $\delta E<10^{-3}$ en 300 s.
\item \textbf{Costo:} el del TP4 crece en forma lineal con $N$ y el del TP3, mucho más rápido: el TP3 gana hasta $N=400$, pero con $N=650$ tarda 10 veces más.
\item \textbf{Densidad:} con $x_0=r$, $\langle t_{90}\rangle$ y $\langle t_{100}\rangle$ crecen con $N$: no hay una densidad óptima. Con $N=650$ las partículas quedan casi en una red y en 300 s solo se usa el 5\,\%.
\item \textbf{Posición:} los extremos son los peores. Con $N$ de 50 a 200 el óptimo está en $x_0\approx0{,}16$--$0{,}26$ m; con $N=20$, el mínimo es una meseta entre 0,11 y 0,40 m.
\item \textbf{Rapidez:} en $\approx1$ s se llega a Maxwell--Boltzmann. $k_BT$ y $m\langle v^2\rangle/2$ difieren en un 0,8\,\%, del orden del desvío entre realizaciones; la diferencia con $m v_0^2/2$ es la energía de los contactos.
\end{itemize}
\end{frame}

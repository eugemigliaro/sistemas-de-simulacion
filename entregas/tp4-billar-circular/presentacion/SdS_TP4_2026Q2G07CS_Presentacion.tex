\node at (0,.95) {$i$};\node at (1.15,.95) {$j$};

\begin{column}{.54\textwidth}
\textbf{Segunda ley de Newton}
\[m_i\ddot{\mathbf r}_i=\sum_j\mathbf F_{ij}\]
\textbf{Modelo normal elástico}
\[\xi_{ij}=r_i+r_j-\|\mathbf r_j-\mathbf r_i\|\]
\[\mathbf F_{ij}=-k\xi_{ij}\widehat{\mathbf e}_{ij}\quad(\xi_{ij}>0)\]
\begin{itemize}
\item $\widehat{\mathbf e}_{ij}$ apunta de $i$ a $j$.
\item Sin fricción ni disipación.
\end{itemize}

\end{column}\end{columns}

\begin{frame}{Tiempos por realización y promedios entre corridas}
\begin{columns}[T,onlytextwidth]
\begin{column}{.49\textwidth}
\textbf{En cada realización $s$}
\[\fu^{(s)}(t)=\frac1N\sum_{i=1}^N\mathbf1_{\{i\text{ usada en }t\}}\]
\[t_p^{(s)}=\inf\{t:\fu^{(s)}(t)\geq p\}\]
\[p=0{,}9\quad\text{o}\quad p=1\]
\textbf{Primero un tiempo por corrida.}
\end{column}
\begin{column}{.48\textwidth}
\textbf{Entre $n_r$ realizaciones}
\[\overline t_p=\frac1{n_r}\sum_{s=1}^{n_r}t_p^{(s)}\]
\[s_{t_p}=\sqrt{\frac{\sum_s(t_p^{(s)}-\overline t_p)^2}{n_r-1}}\]
\begin{itemize}
\item Barras: desvío estándar entre corridas.
\end{itemize}
\end{column}\end{columns}

\end{frame}

\begin{frame}\sectionpage\end{frame}
% 28 -- 55 s
\begin{frame}{Conclusiones}
\small
\begin{itemize}\setlength\itemsep{.22cm}
\item Beeman da el menor ECM, $\approx13\,\%$ menos que Verlet con el mismo orden. Verlet original y Velocity Verlet coinciden; Euler predictor-corrector es de primer orden.
\item $h=5\times10^{-5}$ s conserva la energía dentro del $0{,}1\,\%$ en $30$ s; para $100$--$600$ s se usó $h=1{,}25\times10^{-5}$ s.
\item Al duplicar $N$, paso temporal tarda $\approx\times2{,}5$ y eventos $\times10$--$19$: eventos es más rápido hasta $N=200$ y paso temporal desde $N=400$; con $N=650$, eventos tarda $\times35$.
\item Existe una posición de menor $\langle t_{90}\rangle$ y depende de la frecuencia de choques: $0{,}16$ m con $N=100$ y $0{,}40$ m con $N=20$.
\item La rapidez relaja a Maxwell--Boltzmann con $k_BT=1{,}240\times10^{-2}$ J, próximo a $m\langle v^2\rangle/2$; el resto de $mv_0^2/2$ es energía de contacto.
\item La densidad óptima es baja: $\langle t_{90}\rangle$ es mínimo entre $N=20$ y $100$ y crece desde $N=200$; con $N=650$ la conversión casi se detiene.
\end{itemize}
\end{frame}

\begin{frame}{Fundamentos}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \[
        \bm{x}_i(t+\Delta t)=\bm{x}_i(t)+v\,\Delta t\,
        \bigl(\cos\theta_i(t),\sin\theta_i(t)\bigr)
      \]
      \[
        \theta_i(t+\Delta t)=\Theta_i(t)+\Delta\theta_i(t)
      \]
    \end{column}
    \begin{column}{0.38\textwidth}
      \begin{block}{Ruido normalizado}
        \[
          \eta\in[0,1],\quad
          \Delta\theta_i\sim\mathcal U[-\eta\pi,\eta\pi]
        \]
      \end{block}
      \begin{itemize}
        \item Rapidez constante.
        \item La regla sólo cambia $\Theta_i$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% 5
\begin{frame}{Fundamentos}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Vicsek: promedia}
        \[
          \Theta_i=\operatorname{atan2}\!\left(
          \sum_{j\in\mathcal N_i}\sin\theta_j,
          \sum_{j\in\mathcal N_i}\cos\theta_j\right)
        \]
        Usa todas las partículas del vecindario, incluida la propia.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Votante: copia}
        \[
          J_i\sim\mathcal U(\mathcal N_i),\qquad
          \Theta_i=\theta_{J_i}
        \]
        Elige una única partícula del vecindario, incluida la propia.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \centering\scriptsize Vicsek, PRL, 1995 \quad $\cdot$ \quad Loscar et al., PRE, 2021
\end{frame}

\begin{frame}{Algoritmo de simulación}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.68\textwidth}
      \begin{beamercolorbox}[sep=0.6em,rounded=true]{block body}
        {\small\ttfamily
        estado $\leftarrow$ inicializar(configuración, semilla)\\[0.25em]
        \textbf{para} paso $=0,\ldots,T$:\\
        \quad vecinos $\leftarrow$ CIM(estado, $r_c$, $M$)\\
        \quad $(v_a,S) \leftarrow$ medir(estado, vecinos)\\
        \quad \textbf{si} paso $=T$: terminar\\
        \quad siguiente $\leftarrow$ copia(estado)\\
        \quad \textbf{para cada} partícula $i$:\\
        \qquad base $\leftarrow$ regla(modelo, estado, vecinos[$i$])\\
        \qquad siguiente[$i$].posición $\leftarrow$\\
        \qquad\quad periódica(posición[$i$] $+v\Delta t\,\hat{e}(\theta_i)$)\\
        \qquad siguiente[$i$].ángulo $\leftarrow$\\
        \qquad\quad normalizar(base $+\mathcal U[-\eta\pi,\eta\pi]$)\\
        \quad estado $\leftarrow$ siguiente
        }
      \end{beamercolorbox}
    \end{column}
    \begin{column}{0.28\textwidth}
      \begin{block}{Sincronía}
        El bucle interno siempre lee el estado en $t$ y escribe en una copia.
      \end{block}
      \begin{block}{Validaciones}
        \begin{itemize}
          \item CIM = fuerza bruta.
          \item $\eta=0$, alineado: $v_a=1$.
          \item Repetible por semilla.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% 25
\begin{frame}{Conclusiones}
  \begin{itemize}
    \item Vicsek tolera entre 7,7 y 15,9 veces más ruido que el votante.
    \item Al aumentar la densidad, Vicsek resiste más ruido y el votante menos.
    \item Para $\rho=2,4,8$, $S>0{,}90$: orden y conectividad pueden desacoplarse.
    \item En baja densidad, la conectividad limita el orden global.
    \item El CIM sostuvo 16-20 ns por evaluación de distancia en este sistema.
  \end{itemize}
\end{frame}

\section{Introducción}
\begin{frame}\sectionpage\end{frame}

\begin{frame}{El orden global puede emerger de reglas locales}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.59\textwidth}
      \includegraphics[width=\textwidth,height=5.2cm,keepaspectratio]{\presfig/sistema-real-bandada.png}
      \vspace{-0.2em}

      {\tiny Ilustración generada para esta presentación.}
    \end{column}
    \begin{column}{0.37\textwidth}
      \begin{itemize}
        \item Cada agente se autopropulsa sin líder ni control central.
        \item La interacción es local: solo importa el vecindario inmediato.
        \item El ruido compite con el alineamiento y controla el orden colectivo.
      \end{itemize}
      \vspace{0.5em}
      \alert{Pregunta:} ¿qué cambia cuando se promedian todas las vecinas o se copia solo una?
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Una dinámica común describe posiciones y orientaciones}
  Cada partícula puntual $i$ tiene posición $\bm{x}_i$, orientación $\theta_i$ y rapidez constante $v$:
  \begin{align*}
    \bm{x}_i(t+1) &= \bm{x}_i(t)+v\,(\cos\theta_i(t),\,\sin\theta_i(t)),\\[0.5em]
    \theta_i(t+1) &= \Theta_i(t)+\Delta\theta_i(t).
  \end{align*}
  En este trabajo el ruido está normalizado como
  \[
    \eta\in[0,1], \qquad \Delta\theta_i\sim\mathcal{U}[-\eta\pi,\eta\pi].
  \]
  La única diferencia entre modelos está en la dirección de interacción $\Theta_i$.
\end{frame}

\begin{frame}{Promediar y copiar usan distinta cantidad de información}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \textbf{Vicsek: promedio vectorial local}
      \[
        \Theta_i=\operatorname{atan2}\!\left(
        \sum_{j\in\mathcal N_i}\sin\theta_j,
        \sum_{j\in\mathcal N_i}\cos\theta_j\right)
      \]
      Usa todas las vecinas dentro de $r_c$, incluida la propia partícula.

      {\scriptsize Vicsek \emph{et al.}, \emph{Phys. Rev. Lett.}, 1995.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Votante: copia de una vecina}
      \[
        J_i\sim\mathcal U(\mathcal N_i),\qquad \Theta_i=\theta_{J_i}
      \]
      Elige una única candidata al azar, también pudiendo elegirse a sí misma.

      {\scriptsize Loscar, Baglietto y Vazquez, \emph{Phys. Rev. E}, 2021.}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \centering\alert{Vicsek atenúa fluctuaciones locales; el votante las transmite sin promediarlas.}
\end{frame}

\section{Implementación}
\begin{frame}\sectionpage\end{frame}

\begin{frame}{El motor separa geometría, dinámica y medición}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{description}
        \item[Estado y dinámica] \texttt{model.hpp}, \texttt{simulation.cpp}
        \item[Geometría periódica] \texttt{geometry.cpp}
        \item[Vecindarios CIM] \texttt{neighbors.cpp}
        \item[Observables] \texttt{observables.cpp}
        \item[Configuración y salida] \texttt{runner.cpp}, \texttt{io.cpp}
      \end{description}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Decisiones de diseño}
      \begin{itemize}
        \item Motor reproducible en C++20.
        \item Regla de interacción intercambiable.
        \item Actualización sincrónica: todo $t+1$ lee únicamente el estado en $t$.
        \item Trayectorias y observables se escriben como texto.
        \item Animación y análisis se ejecutan después, de forma independiente.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Cada paso actualiza todas las partículas en paralelo}
  \begin{enumerate}
    \item Construir la grilla del Cell Index Method con las posiciones en $t$.
    \item Medir $v_a(t)$ y la componente gigante $S(t)$.
    \item Buscar vecinas en la celda propia y las ocho adyacentes, usando imagen mínima.
    \item Calcular $\Theta_i(t)$ con la regla de Vicsek o del votante.
    \item Agregar ruido, avanzar la posición y aplicar el contorno periódico.
    \item Reemplazar el estado completo solo después de calcular todas las partículas.
  \end{enumerate}
  \vspace{0.4em}
  \begin{block}{Validaciones}
    El CIM coincide exactamente con fuerza bruta; con $\eta=0$, el estado alineado satisface $v_a=1$ y permanece absorbente.
  \end{block}
\end{frame}

\begin{frame}{Dos observables separan orden y conectividad}
  \textbf{Polarización instantánea}
  \[
    v_a(t)=\frac{1}{N}\left|\sum_{i=1}^{N}(\cos\theta_i(t),\sin\theta_i(t))\right|\in[0,1].
  \]
  \textbf{Fracción de la componente gigante}
  \[
    S(t)=\frac{1}{N}\max_{c\in\mathcal C(t)}|c|\in(0,1].
  \]
  $\mathcal C(t)$ contiene las componentes conexas de la red geométrica definida por distancias periódicas menores o iguales que $r_c$.

  \vspace{0.5em}
  Para cada semilla se promedia $v_a(t)$ o $S(t)$ en $[4\,000,10\,000]$; cada punto es la media de 10 realizaciones y su barra es la desviación estándar entre realizaciones.
\end{frame}

\section{Conclusiones}
\begin{frame}\sectionpage\end{frame}

\begin{frame}{Las dos reglas producen transiciones cualitativamente distintas}
  \begin{itemize}
    \item Vicsek tolera entre \alert{7,7 y 15,9 veces más ruido} que el votante antes de alcanzar $\langle v_a\rangle=0{,}5$.
    \item Al aumentar $\rho$ de 2 a 8, $\eta_{1/2}$ \alert{crece de 0,470 a 0,555 en Vicsek}, pero \alert{cae de 0,061 a 0,035 en el votante}.
    \item Para $\rho=2,4,8$, la componente gigante permanece por encima de 0,90 y no discrimina estados de orden.
    \item En densidades bajas, $S$ recorre casi todo su rango y crece monótonamente con $v_a$; allí conectividad y orden dejan de ser independientes.
    \item Normalizado por evaluación de distancia, el CIM del TP2 requiere 16--20 ns frente a 35--44 ns en el TP1, bajo geometrías diferentes.
  \end{itemize}
\end{frame}

\begin{frame}{Tiempo de choque entre dos partículas}
  Con $\bm{r}_i(t)=\bm{r}_i(0)+\bm{v}_i t$, el contacto ocurre cuando
  $\lVert\bm{r}_j(t)-\bm{r}_i(t)\rVert=\sigma$, con $\sigma=R_i+R_j$:
  \[
    t_c=\begin{cases}
      \infty & \text{si } \Delta\bm{v}\cdot\Delta\bm{r}\geq 0,\\[2pt]
      \infty & \text{si } d<0,\\[2pt]
      -\dfrac{\Delta\bm{v}\cdot\Delta\bm{r}+\sqrt{d}}{\Delta\bm{v}\cdot\Delta\bm{v}} & \text{en otro caso,}
    \end{cases}
    \qquad
    d=(\Delta\bm{v}\cdot\Delta\bm{r})^2-(\Delta\bm{v}\cdot\Delta\bm{v})\,(\Delta\bm{r}\cdot\Delta\bm{r}-\sigma^2),
  \]
  \[
    \Delta\bm{r}=\bm{r}_j-\bm{r}_i,\qquad \Delta\bm{v}=\bm{v}_j-\bm{v}_i.
  \]
  \vfill
  \begin{block}{Contra una pared vertical}
    $t_c=(x_p-R-x(0))/v_x$ si $v_x>0$, y análogo para las demás paredes.
  \end{block}
\end{frame}

% 5
\begin{frame}{Choque elástico y próximo evento}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{block}{Velocidades después del choque}
        \[
          J=\frac{2m_im_j\,(\Delta\bm{v}\cdot\Delta\bm{r})}{\sigma\,(m_i+m_j)},\qquad
          \bm{J}=\frac{J\,\Delta\bm{r}}{\sigma},
        \]
        \[
          \bm{v}_i^{\,d}=\bm{v}_i^{\,a}+\frac{\bm{J}}{m_i},\qquad
          \bm{v}_j^{\,d}=\bm{v}_j^{\,a}-\frac{\bm{J}}{m_j}.
        \]
        Un obstáculo fijo es el límite $m_j\to\infty$.
      \end{block}
    \end{column}
    \begin{column}{0.41\textwidth}
      \begin{block}{Próximo evento}
        \[
          t^{*}=\min_{\text{pares y paredes}} t_c
        \]
        Todas las partículas avanzan hasta $t^{*}$ y solo cambian las velocidades de las que chocaron.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Algoritmo de simulación}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.67\textwidth}
      \begin{beamercolorbox}[sep=0.6em,rounded=true]{block body}
        {\small\ttfamily
        estado $\leftarrow$ generar(configuración, semilla)\\[0.25em]
        cola $\leftarrow$ agendar todos los choques\\
        \textbf{mientras} cola no esté vacía:\\
        \quad evento $\leftarrow$ extraer mínimo\\
        \quad \textbf{si} evento está invalidado: continuar\\
        \quad \textbf{si} evento.t $>$ tmax: terminar\\
        \quad avanzar todas las partículas hasta evento.t\\
        \quad resolver(evento)\\
        \quad escribir estado si corresponde\\
        \quad reagendar participantes
        }
      \end{beamercolorbox}
    \end{column}
    \begin{column}{0.29\textwidth}
      \begin{block}{Invalidación perezosa}
        Un evento se invalida si cambió el contador de choques de algún participante.
      \end{block}
      \begin{block}{Costo}
        Cola: $O(N\log E)$ por evento; predomina el costo de reagendar participantes.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Goles y tiempo característico}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \textbf{Fracción de partículas usadas}
      \[
        F_u^{(s)}(t)=\frac{N_g^{(s)}(t)}{N}
      \]
      \textbf{Tiempo por realización $s$}
      \[
        t_{90}^{(s)}=\min\{t:F_u^{(s)}(t)\geq0{,}9\}
      \]
      \textbf{Promedio sobre $n_r$ realizaciones}
      \[
        \langle t_{90}\rangle=\frac{1}{n_r}\sum_{s=1}^{n_r}t_{90}^{(s)}
      \]
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Repeticiones}
        {\small
        Disco central: 20 por radio.\\
        Profundidad: 24 por valor.\\
        Comparación final: 100.\\
        $t_{\max}=60$ a $100\ \mathrm{s}$; barras: error estándar.}
      \end{block}
      \small Las corridas que no llegan al 90\,\% antes de $t_{\max}$ se informan, no se descartan.
    \end{column}
  \end{columns}
\end{frame}

% 14
\begin{frame}{Difusión}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.51\textwidth}
      \textbf{Desplazamiento cuadrático medio}
      \[
        \mathrm{DCM}^{(s)}(t)=\frac{1}{N}\sum_{i=1}^{N}
        \left\lVert\bm{x}_i^{(s)}(t)-\bm{x}_i^{(s)}(0)\right\rVert^2
      \]
      En la ventana difusiva:
      \[
        \mathrm{DCM}(t)=4Dt
      \]
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Ajuste por realización}
        \[
          E(c)=\sum_j\left[\mathrm{DCM}(t_j)-c\,t_j\right]^2,\qquad
          D=\frac{c_{\min}}{4}
        \]
      \end{block}
      \begin{block}{Protocolo}
        10 realizaciones de $8\ \mathrm{s}$ por configuración; ventana común $[0{,}3;1{,}5]\ \mathrm{s}$; se informa $\langle D\rangle\pm\sigma_D$.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusiones}
  \begin{itemize}
    \item El tiempo de ejecución crece con $N$ y se dispara al acercarse al empaquetamiento compacto: con $N=675$ tarda $9{,}5$ min y con $N=700$ ya no termina en 10 min.
    \item Un disco central reduce $\langle t_{90}\rangle$ a medida que crece su radio: de $22{,}9$ a $15{,}3\ \mathrm{s}$.
    \item Separar las mitades casi no ayuda; lo que reduce $t_{90}$ es quitar área lejos de los arcos sin angostar la entrada.
    \item La superelipse elegida da $\langle t_{90}\rangle=13{,}61\pm0{,}17\ \mathrm{s}$: un 40\,\% menos que la mesa vacía y un 13\,\% menos que el mejor disco central.
    \item $D$ y $t_{90}$ no tienen una relación única: la correlación cambia de signo entre familias.
  \end{itemize}
\end{frame}
